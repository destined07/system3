def fibo(n):
    if n <= 2:
        return n - 1
    return fibo(n - 1) + fibo(n - 2)


n = int(input())

print("USING ITERATIVE")

if n == 1:
    print(0, end=" ")
if n == 2:
    print(1, end=" ")

p = 1
pp = 0

for i in range(3, n + 1):
    curr = p + pp
    pp = p
    p = curr
    if i == n:
        print(curr, end=" ")

print("USING RECURSIVE")
print(fibo(n))
