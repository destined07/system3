items = []

mxwg = float(input())
n = int(input())

for i in range(1, n + 1):
    price = int(input(f"Enter the price of {i}th item- "))
    wg = int(input(f"Enter the wg of {i}th item- "))

    items.append((price, wg))


items.sort(key=lambda x: x[0] / x[1], reverse=True)

remwg = mxwg
profit = 0

for i in range(n):
    curr_price = items[i][0]
    curr_wg = items[i][1]

    if remwg >= curr_wg:
        profit += curr_price
        remwg -= curr_wg
    else:
        profit += (curr_price / curr_wg) * remwg
        remwg = 0

print(profit)
