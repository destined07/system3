res = []


def nqueen(g, cols, diffs, sums, row):
    n = len(g)

    if row == n:
        res.append([r[:] for r in g])
        return

    for c in range(n):
        if c not in cols and (row - c) not in diffs and (row + c) not in sums:

            cols.add(c)
            diffs.add(row - c)
            sums.add(row + c)

            g[row][c] = 'Q'

            nqueen(g, cols, diffs, sums, row + 1)

            cols.remove(c)
            diffs.remove(row - c)
            sums.remove(row + c)

            g[row][c] = '.'


def solve(n):
    g = [['.' for j in range(n)] for i in range(n)]

    cols = set()
    diffs = set()
    sums = set()

    nqueen(g, cols, diffs, sums, 0)


n = int(input())

solve(n)

for g in res:
    for r in g:
        for ch in r:
            print(ch, end=" ")
        print()

    print("\n")
